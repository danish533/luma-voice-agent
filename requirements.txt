# Voice pipeline. The extras pull in only the providers this agent uses.
#
# Pinned to the 1.6 line, not `~=1.0`. The results in EVALUATION.md were
# measured on 1.6.7, and a floating major-or-minor pin means a rebuild six
# months from now silently installs 1.9 and reports numbers for software that
# was never run. Turn detection in particular has changed behaviour across
# minor versions here before. Patch releases within 1.6 are still allowed.
livekit-agents[deepgram,openai,google,silero,turn-detector]~=1.6.7

httpx==0.28.1
python-dotenv==1.1.1

# Windows ships no system timezone database, so zoneinfo cannot resolve
# "America/Los_Angeles" and config.py raises at import — every entry point dies
# before it runs. Linux and macOS read the OS database and do not need this.
tzdata; sys_platform == "win32"

# To run the vendored mock reservation API from this same venv. pydantic is
# deliberately unpinned: the starter package pins 2.11.4, which forces an old
# google-genai and breaks the Gemini path. The mock API uses only basic v2
# features and runs fine on current pydantic.
fastapi==0.115.12
uvicorn[standard]==0.34.2

# Test and evaluation
pytest==9.1.1
pytest-asyncio==1.3.0

# --- Production layer (optional; the agent runs without any of it) ---------
# Persists the conversation -- the data the reservation service does not own.
sqlalchemy[asyncio]>=2.0
asyncpg>=0.30          # Postgres in production
aiosqlite>=0.20        # SQLite for local dev and CI

# Schema migrations. Alembic runs synchronously and cannot drive asyncpg, so
# migrations/env.py swaps the async driver for psycopg2 against the same URL.
# Both are needed even though the application itself only ever uses asyncpg.
alembic>=1.13
psycopg2-binary>=2.9

# Shared idempotency and the short-TTL availability cache.
redis>=5.0
fakeredis>=2.23        # in-process Redis for tests

# Ops console auth. Password hashing uses hashlib.scrypt from the standard
# library -- passlib+bcrypt was dropped after passlib 1.7.4's backend probe
# turned out to be incompatible with bcrypt >= 4.1.
itsdangerous>=2.2
python-multipart>=0.0.9   # the sign-in form posts as form data

# Metrics. Arrives transitively with livekit-agents, but src/luma/metrics.py
# imports it directly, and depending on someone else's transitive dependency is
# how a working build breaks on an unrelated upgrade.
prometheus_client>=0.20
